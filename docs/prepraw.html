1.1 从PubChem获取分子信息

rm(list = ls())

# 读取原始成分数据
ingredients_raw <- read.xlsx("data/qfd_ingredients/raw.xlsx") |>
  mutate(PubChem.CID = as.character(PubChem.CID)) |>
  distinct(Abbreviation, PubChem.CID, .keep_all = TRUE) |>
  arrange(Abbreviation)

# 从PubChem获取信息
source("utils.R")

properties <- c("IUPACName", "MolecularFormula", "MolecularWeight", "SMILES")

unique_cids <- unique(ingredients_raw$PubChem.CID)
cid_batches <- split(unique_cids, ceiling(seq_along(unique_cids)/10))

plan(multisession, workers = availableCores() - 2)
with_progress({
  p <- progressor(steps = length(cid_batches))
  chem_info <- future_map_dfr(cid_batches, function(b) {
    p()
    fetch_batch(properties, b)
  })
})
plan(sequential)

ingredients <- ingredients_raw |>
  left_join(chem_info, by = c("PubChem.CID"="CID")) |>
  mutate(across(where(is.character), trimws)) |>
  mutate(MolecularWeight = as.numeric(MolecularWeight)) |>
  arrange(Abbreviation, MolecularWeight)

save(ingredients,file = "data/qfd_ingredients/ingredients.RData")

2.12 合并潜在成分靶点

load("data/qfd_targets/ingredients.std.RData")

ingredients.targets <- rbind(
  pubchem_std,
  stitch_std,
  tcmsp_std,
  herb2_std,
  swiss_std,
  super_std,
  pharm_std,
  sea_std
) |> 
  group_by(No., Abbreviation, Ingredient, Type, PubChem.CID, Symbol) |>
  summarise(
    Source.Sum = paste(Source, collapse = ";"),  
    Class.Sum = paste(Class, collapse = ";"), 
    Count = n_distinct(Source),      
    .groups = "drop"
  ) |>
  mutate(Class = case_when(
    str_detect(Class.Sum, "validated") ~ "validated",
    TRUE ~ "predicted"
  )) |>
  mutate(Symbol = trimws(Symbol))

ingredients.targets.filter <-  ingredients.targets |>
  dplyr::filter(Class == "validated" | (Class == "predicted" & Count>=2))

save(ingredients.targets,ingredients.targets.filter, file =  "data/qfd_targets/ingredients_targets.RData")

  dplyr::filter(!is.na(Gene.Names.primary)) |>
